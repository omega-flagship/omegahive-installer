cases = []
def add(name, doc, mind=M, module=None, stricter=False):
    cases.append({'name': name, 'mind': mind, 'module': module or doc.get('module'), 'doc': doc, 'surface_stricter': stricter})
def cur(m, **kw):
    d = copy.deepcopy(by[m]); d['state'] = 'current'
    d['provenance'] = {'source_revision': 'src-1', 'evidence_refs': [], 'read_path': 'host-consent-filtered-snapshot', 'authority': 'none'}
    d.update(kw); return d
def unav(m, **kw):
    d = {k: by[m][k] for k in ('module', 'mind_id', 'qualification', 'explain')}
    d.update({'revision': None, 'observed_at': None, 'available': False, 'state': 'unavailable', 'unavailable_reason': 'no_reader', 'items': [],
              'provenance': {'source_revision': None, 'evidence_refs': [], 'read_path': 'none', 'authority': 'none'}})
    d.update(kw); return d
